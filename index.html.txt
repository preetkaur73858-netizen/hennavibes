  <script>
    // Base price in smallest unit (here rupee units as integer)
    const basePrice = 25000; // e.g. 25,000 (₹) per night

    // Intl NumberFormat for Indian Rupees
    const inrFormatter = new Intl.NumberFormat('en-IN', {
      style: 'currency',
      currency: 'INR',
      maximumFractionDigits: 0
    });

    const pricePerNightEl = document.getElementById('pricePerNight');
    const totalPriceEl = document.getElementById('totalPrice');
    const guestInput = document.getElementById('guest');

    function renderPrices() {
      const guests = Math.max(1, parseInt(guestInput.value || 1, 10));
      // Example pricing rule: total = basePrice * guests
      const total = basePrice * guests;

      // Update DOM with localized INR strings (uses ₹ symbol)
      pricePerNightEl.textContent = inrFormatter.format(basePrice);
      pricePerNightEl.setAttribute('aria-label', `Price per night ${inrFormatter.format(basePrice)}`);
      totalPriceEl.textContent = inrFormatter.format(total);
      totalPriceEl.setAttribute('aria-label', `Total price ${inrFormatter.format(total)}`);
    }

    // Initialize
    renderPrices();

    // Recalculate when guests change
    guestInput.addEventListener('input', () => {
      renderPrices();
    });

    // Respect reduced motion: avoid heavy transitions when user prefers reduced motion
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
      document.documentElement.style.setProperty('--carousel-transition', 'none');
    }
  </script>
